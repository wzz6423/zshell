// ZSHELL_INTEGRATION_ID=pi
// ZSHELL_INTEGRATION_VERSION=1
// @ts-nocheck

import net from "node:net";
import os from "node:os";
import path from "node:path";

type AgentState = "working" | "blocked" | "idle";

const socketPath = process.env.ZSHELL_AUTOMATION_SOCKET;
const token = process.env.ZSHELL_AUTOMATION_TOKEN;
const terminalID = process.env.ZSHELL_TERMINAL_ID;
let requestSequence = Date.now() * 1000;

function enabled(): boolean {
  return process.env.ZSHELL_AUTOMATION === "1"
    && !!socketPath
    && !!token
    && !!terminalID;
}

function requestOnce(method: string, params): Promise<void> {
  if (!enabled()) return Promise.resolve();

  requestSequence += 1;
  const request = {
    version: 1,
    id: `zshell:pi:${requestSequence}`,
    method,
    token,
    terminalID,
    params,
  };

  return new Promise((resolve) => {
    const socket = net.createConnection(socketPath!);
    let finished = false;
    const finish = () => {
      if (finished) return;
      finished = true;
      socket.destroy();
      resolve();
    };
    socket.setTimeout(750, finish);
    socket.on("connect", () => socket.write(`${JSON.stringify(request)}\n`));
    socket.on("data", finish);
    socket.on("error", finish);
    socket.on("end", finish);
    socket.on("close", finish);
  });
}

export default function zshellAgentState(pi) {
  if (!enabled()) return;

  let rootSession = false;
  let active = false;
  let lastState: AgentState | undefined;
  let requestChain = Promise.resolve();

  function enqueue(method, params) {
    const pending = requestChain.then(() => requestOnce(method, params));
    requestChain = pending.catch(() => {});
    return pending;
  }

  function publish(state: AgentState) {
    if (state === lastState) return;
    lastState = state;
    void enqueue("agent.report", { state, reason: `Pi lifecycle: ${state}` });
  }

  async function bindConversation(event, context) {
    if (context?.mode !== "tui") return;
    rootSession = true;
    active = context?.isIdle?.() === false;
    try {
      const sessionID = context.sessionManager.getSessionId();
      const sessionFile = context.sessionManager.getSessionFile();
      if (sessionFile && /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(sessionID)) {
        await enqueue("agent.session.start", {
          identity: {
            provider: "pi",
            sessionID,
            sessionFile,
            workingDirectory: context.sessionManager.getCwd(),
            configDirectory: path.resolve(process.env.PI_CODING_AGENT_DIR || path.join(os.homedir(), ".pi", "agent")),
          },
          reporter_pid: process.pid,
          source: event?.reason ?? "startup",
        });
      } else {
        await enqueue("agent.session.clear", {
          provider: "pi", reporter_pid: process.pid,
        });
      }
    } catch {
      await enqueue("agent.session.clear", { provider: "pi", reporter_pid: process.pid });
    }
    lastState = undefined;
    publish(active ? "working" : "idle");
  }

  pi.on("session_start", bindConversation);
  // Earlier Pi releases use switch/fork instead of restarting the extension.
  pi.on("session_switch", bindConversation);
  pi.on("session_fork", bindConversation);

  pi.on("agent_start", () => {
    if (!rootSession) return;
    active = true;
    publish("working");
  });

  const settle = (_event, context) => {
    if (!rootSession || context?.isIdle?.() === false) return;
    active = false;
    publish("idle");
  };
  pi.on("agent_settled", settle);
  // Older Pi releases expose agent_end but not agent_settled.
  pi.on("agent_end", settle);
}
